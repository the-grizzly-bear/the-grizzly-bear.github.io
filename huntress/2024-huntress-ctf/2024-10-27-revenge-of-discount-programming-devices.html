import re
import binascii
import codecs

def decode_hex_and_binary(input_file, output_file):
    """
    Decode strace output containing hex and binary data into readable format
    with improved handling of binary data
    """
    def hex_dump(data):
        """
        Convert binary data to a readable hex dump format with ASCII representation
        """
        try:
            # Remove quotes if present
            data = data.strip('"')
            # Convert escaped sequences
            try:
                data = codecs.escape_decode(bytes(data, "utf-8"))[0].decode('latin1')
            except:
                pass
            
            # Convert to hex representation
            hex_str = ' '.join(f'{ord(c):02x}' for c in data[:16])  # Show first 16 bytes only
            
            # Create ASCII representation (show . for non-printable characters)
            ascii_str = ''.join(c if 32 <= ord(c) <= 126 else '.' for c in data[:16])
            
            if len(data) > 16:
                return f"[{hex_str}] ({ascii_str})... +{len(data)-16} bytes"
            return f"[{hex_str}] ({ascii_str})"
        except:
            return data

    def process_line(line):
        # Handle write/read calls with binary data
        write_pattern = r'write\((\d+), "(.*?)"(?:\.{3})?, (\d+)\) = (\d+)'
        read_pattern = r'read\((\d+), "(.*?)"(?:\.{3})?, (\d+)\) = (\d+)'
        
        # Process write calls
        if 'write(' in line:
            match = re.match(write_pattern, line)
            if match:
                fd, data, size, result = match.groups()
                decoded = hex_dump(data)
                return f'write(fd={fd}, data={decoded}, size={size}) = {result}'
        
        # Process read calls
        if 'read(' in line:
            match = re.match(read_pattern, line)
            if match:
                fd, data, size, result = match.groups()
                decoded = hex_dump(data)
                return f'read(fd={fd}, data={decoded}, size={size}) = {result}'
        
        # Decode hex addresses
        line = re.sub(r'0x([0-9a-fA-F]+)', lambda m: f'0x{m.group(1)} (Addr)', line)
        
        # Handle other string patterns
        def decode_string(match):
            try:
                s = match.group(1)
                if all(32 <= ord(c) <= 126 for c in s):
                    return f'"{s}"'
                return f'{hex_dump(s)}'
            except:
                return match.group(0)
        
        line = re.sub(r'"([^"]*)"', decode_string, line)
        return line

    with open(input_file, 'r') as f_in, open(output_file, 'w') as f_out:
        f_out.write("# Decoded strace output\n")
        f_out.write("# Format for binary data: [hex bytes] (ASCII representation)... +remaining bytes\n")
        f_out.write("# '.' in ASCII representation indicates non-printable character\n\n")
        
        for line in f_in:
            # Process the line
            decoded_line = process_line(line.strip())
            
            # Add explanatory comments for common syscalls
            if 'execve(' in line:
                decoded_line += ' # Execute program'
            elif 'brk(' in line:
                decoded_line += ' # Change data segment size'
            elif 'mmap(' in line:
                decoded_line += ' # Map memory'
            elif 'openat(' in line:
                decoded_line += ' # Open file'
            elif 'read(' in line:
                decoded_line += ' # Read from file descriptor'
            elif 'write(' in line:
                decoded_line += ' # Write to file descriptor'
            elif 'fstat(' in line:
                decoded_line += ' # Get file status'
            elif 'close(' in line:
                decoded_line += ' # Close file descriptor'
            
            f_out.write(decoded_line + '\n')

import re

def extract_elf_files(strace_file, output_dir='extracted_files'):
    """Extract ELF files from strace output by following write operations."""
    
    # Ensure the output directory exists
    Path(output_dir).mkdir(parents=True, exist_ok=True)
    
    # Dictionary to keep track of file descriptors and their data
    fd_to_file = {}
    fd_data = {}
    extracted_files = []
    
    # Regex patterns
    # Updated to capture both 'write(fd=3, data=[...])' and 'write(3, data=[...])'
    open_pattern = re.compile(r'openat\(AT_FDCWD, "([^"]+)".*\) = (\d+)')
    write_pattern_bracket_fd = re.compile(r'write\(fd=(\d+), data=\[([0-9a-fA-F ]+)\]')
    write_pattern_bracket = re.compile(r'write\((\d+), data=\[([0-9a-fA-F ]+)\]')
    write_pattern_escaped_fd = re.compile(r'write\(fd=(\d+), "\\x([0-9a-fA-F]{2}(?:\\x[0-9a-fA-F]{2})*)", (\d+)\)')
    write_pattern_escaped = re.compile(r'write\((\d+), "\\x([0-9a-fA-F]{2}(?:\\x[0-9a-fA-F]{2})*)", (\d+)\)')
    close_pattern = re.compile(r'close\((\d+)\)')
    
    with open(strace_file, 'r') as f:
        for line in f:
            # Detect file open operations
            open_match = open_pattern.search(line)
            if open_match:
                filename, fd = open_match.groups()
                fd_to_file[fd] = filename
                fd_data[fd] = bytearray()
                print(f"Opened file: {filename} (fd={fd})")
                continue
            
            # Detect write operations with data in brackets and 'fd='
            write_bracket_fd_match = write_pattern_bracket_fd.search(line)
            if write_bracket_fd_match:
                fd, hex_str = write_bracket_fd_match.groups()
                try:
                    binary_data = binascii.unhexlify(hex_str.replace(' ', ''))
                    if fd in fd_to_file:
                        fd_data[fd].extend(binary_data)
                        print(f"Added {len(binary_data)} bytes to {fd_to_file[fd]} (fd={fd})")
                except binascii.Error as e:
                    print(f"Error decoding hex data in line: {line.strip()} | Error: {e}")
                continue
            
            # Detect write operations with data in brackets without 'fd='
            write_bracket_match = write_pattern_bracket.search(line)
            if write_bracket_match:
                fd, hex_str = write_bracket_match.groups()
                try:
                    binary_data = binascii.unhexlify(hex_str.replace(' ', ''))
                    if fd in fd_to_file:
                        fd_data[fd].extend(binary_data)
                        print(f"Added {len(binary_data)} bytes to {fd_to_file[fd]} (fd={fd})")
                except binascii.Error as e:
                    print(f"Error decoding hex data in line: {line.strip()} | Error: {e}")
                continue
            
            # Detect write operations with escaped hex data and 'fd='
            write_escaped_fd_match = write_pattern_escaped_fd.search(line)
            if write_escaped_fd_match:
                fd, hex_str, size = write_escaped_fd_match.groups()
                try:
                    binary_data = binascii.unhexlify(hex_str)
                    if fd in fd_to_file:
                        fd_data[fd].extend(binary_data)
                        print(f"Added {len(binary_data)} bytes to {fd_to_file[fd]} (fd={fd})")
                except binascii.Error as e:
                    print(f"Error decoding escaped hex data in line: {line.strip()} | Error: {e}")
                continue
            
            # Detect write operations with escaped hex data without 'fd='
            write_escaped_match = write_pattern_escaped.search(line)
            if write_escaped_match:
                fd, hex_str, size = write_escaped_match.groups()
                try:
                    binary_data = binascii.unhexlify(hex_str)
                    if fd in fd_to_file:
                        fd_data[fd].extend(binary_data)
                        print(f"Added {len(binary_data)} bytes to {fd_to_file[fd]} (fd={fd})")
                except binascii.Error as e:
                    print(f"Error decoding escaped hex data in line: {line.strip()} | Error: {e}")
                continue
            
            # Detect close operations
            close_match = close_pattern.search(line)
            if close_match:
                fd = close_match.group(1)
                if fd in fd_to_file:
                    filename = fd_to_file.pop(fd)
                    data = fd_data.pop(fd, bytearray())
                    # Check for ELF magic bytes
                    #if data.startswith(b'\x7fELF'):
                    #    output_path = Path(output_dir) / Path(filename).name
                    ##    with open(output_path, 'wb') as out_file:
                    #        out_file.write(data)
                    #    extracted_files.append(str(output_path))
                    #    print(f"Saved ELF file: {output_path} ({len(data)} bytes)")
                    #else:
                    #    print(f"Data written to {filename} does not start with ELF magic bytes. Skipping.")
                    # Always save the file, regardless of magic bytes
                    if data:
                        output_path = Path(output_dir) / Path(filename).name
                        with open(output_path, 'wb') as out_file:
                            out_file.write(data)
                        extracted_files.append(str(output_path))
                        print(f"Saved file: {output_path} ({len(data)} bytes)")
                continue
    
    return extracted_files

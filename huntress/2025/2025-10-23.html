cat extract_zip.py
#!/usr/bin/env python3
import re

# Read the netsupport file
with open('netsupport', 'r', encoding='utf-8-sig') as f:
    content = f.read()

# Find the byte array
match = re.search(r'\[Byte\[\]\]\$\w+\s*=\s*@\(([\d,\s]+)\)', content)

if match:
    # Extract the bytes
    byte_str = match.group(1)
    bytes_list = [int(x.strip()) for x in byte_str.split(',') if x.strip()]

    # Convert to bytes and write to file
    byte_data = bytes(bytes_list)

    with open('extracted.zip', 'wb') as f:
        f.write(byte_data)

    print(f"Extracted {len(byte_data)} bytes to extracted.zip")
    print(f"First bytes: {byte_data[:20].hex()}")
else:
    print("No byte array found")

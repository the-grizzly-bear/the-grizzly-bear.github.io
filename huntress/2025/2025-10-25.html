-- The main decoding function
local function __D(t, k)
local T = __T() -- Get the character set
-- Simple string hashing function
local S = function(k)
k = tostring(k or "")
local s = 3237919422
for i = 1, #k do
s = (s * 1315423911 + k:byte(i)) % 4294967296
end
return s
end
-- Pseudo-random number generator (PRNG) based on seed
local M = function(seed)
return function()
seed = (1664525 * seed + 1013904223) % 4294967296
return seed
end
end
-- Simple XOR function for 4 bits
local X = function(a, b)
local r, p = 0, 1
for _ = 1, 4 do
local A = a % 2
local B = b % 2
if (A + B) % 2 == 1 then r = r + p end
a = (a - A) / 2
b = (b - B) / 2
p = p * 2
end
return r
end

local perm = {} -- Permutation table
for n = 0, 15 do perm[n] = n end

local r = M(S(tostring(k) .. "|perm"))

lo = X(lo, kl)
end
out[#out + 1] = string.char(hi * 16 + lo) -- Combine nibbles and add character to output
i = i + step -- Move to next pair of sequences
end
return table.concat(out) -- Return the final decoded string
end
